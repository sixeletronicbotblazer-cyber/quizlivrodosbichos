    40→type PixelWindow = Window & {
    41→  fbq?: (...args: unknown[]) => void;
    42→  gtag?: (...args: unknown[]) => void;
    43→  dataLayer?: unknown[];
    44→};
    45→
    46→const firedEvents = new Set<string>();
    47→
    48→function trackEvent(name: string, params?: Record<string, unknown>) {
    49→  if (typeof window === "undefined" || firedEvents.has(name)) return;
    50→  firedEvents.add(name);
    51→  try {
    52→    const w = window as PixelWindow;
    53→    if (typeof w.fbq === "function") w.fbq("trackCustom", name, params);
    54→    if (typeof w.gtag === "function") w.gtag("event", name, params);
    55→    if (Array.isArray(w.dataLayer)) w.dataLayer.push({ event: name, ...params });
    56→  } catch {
    57→    /* pixel ausente ou bloqueado: ignora silenciosamente */
    58→  }
    59→}

   177→
   178→type Stage = "intro" | "quiz" | "result";
   179→const STORAGE_KEY = "livro-dos-sonhos-quiz-v1";
   180→
   181→type QuizState = { stage: Stage; current: number; answers: (string | null)[] };
   182→
   183→const defaultState: QuizState = {
   184→  stage: "intro",
   185→  current: 0,
   186→  answers: Array(QUESTIONS.length).fill(null),
   187→};
   188→
   189→function isValidState(data: unknown): data is QuizState {
   190→  if (typeof data !== "object" || data === null) return false;
   191→  const d = data as Partial<QuizState>;
   192→  const validAnswers =
   193→    Array.isArray(d.answers) &&
   194→    d.answers.length === QUESTIONS.length &&
   195→    d.answers.every((a) => a === null || typeof a === "string");
   196→  if (!validAnswers) return false;
   197→  if (d.stage === "quiz") {
   198→    return typeof d.current === "number" && d.current >= 0 && d.current < QUESTIONS.length;
   199→  }
   200→  if (d.stage === "result") {
   201→    return (d.answers as (string | null)[]).every((a) => a !== null);
   202→  }
   203→  return false;
   204→}
   205→
   206→/** Lê as respostas salvas na sessão (uma única vez). */
   207→function readSavedState(): QuizState | null {
   208→  try {
   209→    const raw = sessionStorage.getItem(STORAGE_KEY);
   210→    if (!raw) return null;
   211→    const data = JSON.parse(raw);
   212→    return isValidState(data) ? data : null;
   213→  } catch {
   214→    return null;
   215→  }
   216→}
   217→
   218→/* Mini armazenamento externo: fonte única de verdade do progresso.
   219→   Mantém as respostas salvas localmente durante a sessão (reload não
   220→   perde o progresso) sem quebrar a hidratação do servidor. */
   221→let memoryState: QuizState = defaultState;
   222→let hydratedFromStorage = false;
   223→const listeners = new Set<() => void>();
   224→
   225→function getSnapshot(): QuizState {
   226→  if (!hydratedFromStorage) {
   227→    hydratedFromStorage = true;
   228→    memoryState = readSavedState() ?? memoryState;
   229→  }
   230→  return memoryState;
   231→}
   232→
   233→function getServerSnapshot(): QuizState {
   234→  return defaultState;
   235→}
   236→
   237→function subscribe(listener: () => void): () => void {
   238→  listeners.add(listener);
   239→  return () => {
   240→    listeners.delete(listener);
   241→  };
   242→}
   243→
   244→function updateQuizState(next: QuizState): void {
   245→  memoryState = next;
   246→  try {
   247→    if (next.stage === "intro") {
   248→      sessionStorage.removeItem(STORAGE_KEY);
   249→    } else {
   250→      /* mantém quiz em andamento E resultado concluído salvos
   251→         durante a sessão (recarregar não perde o progresso) */
   252→      sessionStorage.setItem(STORAGE_KEY, JSON.stringify(next));
   253→    }
   254→  } catch {
   255→    /* armazenamento indisponível: mantém apenas em memória */
   256→  }
   257→  listeners.forEach((listener) => listener());
   258→}
