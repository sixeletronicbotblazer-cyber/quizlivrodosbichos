    40→    40→type PixelWindow = Window & {

    47→    47→
    48→    48→function trackEvent(name: string, params?: Record<string, unknown>) {
    49→    49→  if (typeof window === "undefined" || firedEvents.has(name)) return;
    50→    50→  firedEvents.add(name);
    51→    51→  try {
    52→    52→    const w = window as PixelWindow;
    53→    53→    if (typeof w.fbq === "function") w.fbq("trackCustom", name, params);
    54→    54→    if (typeof w.gtag === "function") w.gtag("event", name, params);
    55→    55→    if (Array.isArray(w.dataLayer)) w.dataLayer.push({ event: name, ...params });
    56→    56→  } catch {
    57→    57→    /* pixel ausente ou bloqueado: ignora silenciosamente */
    58→    58→  }
    59→    59→}

   181→   181→type QuizState = { stage: Stage; current: number; answers: (string | null)[] };

   188→   188→
   189→   189→function isValidState(data: unknown): data is QuizState {
   190→   190→  if (typeof data !== "object" || data === null) return false;
   191→   191→  const d = data as Partial<QuizState>;
   192→   192→  const validAnswers =
   193→   193→    Array.isArray(d.answers) &&
   194→   194→    d.answers.length === QUESTIONS.length &&
   195→   195→    d.answers.every((a) => a === null || typeof a === "string");
   196→   196→  if (!validAnswers) return false;
   197→   197→  if (d.stage === "quiz") {
   198→   198→    return typeof d.current === "number" && d.current >= 0 && d.current < QUESTIONS.length;
   199→   199→  }
   200→   200→  if (d.stage === "result") {
   201→   201→    return (d.answers as (string | null)[]).every((a) => a !== null);
   202→   202→  }
   203→   203→  return false;
   204→   204→}

   206→   206→/** Lê as respostas salvas na sessão (uma única vez). */
   207→   207→function readSavedState(): QuizState | null {
   208→   208→  try {
   209→   209→    const raw = sessionStorage.getItem(STORAGE_KEY);
   210→   210→    if (!raw) return null;
   211→   211→    const data = JSON.parse(raw);
   212→   212→    return isValidState(data) ? data : null;
   213→   213→  } catch {
   214→   214→    return null;
   215→   215→  }
   216→   216→}
   217→   217→
   218→   218→/* Mini armazenamento externo: fonte única de verdade do progresso.
   219→   219→   Mantém as respostas salvas localmente durante a sessão (reload não
   220→   220→   perde o progresso) sem quebrar a hidratação do servidor. */
   221→   221→let memoryState: QuizState = defaultState;
   222→   222→let hydratedFromStorage = false;
   223→   223→const listeners = new Set<() => void>();
   224→   224→
   225→   225→function getSnapshot(): QuizState {
   226→   226→  if (!hydratedFromStorage) {
   227→   227→    hydratedFromStorage = true;
   228→   228→    memoryState = readSavedState() ?? memoryState;
   229→   229→  }
   230→   230→  return memoryState;
   231→   231→}
   232→   232→
   233→   233→function getServerSnapshot(): QuizState {
   234→   234→  return defaultState;
   235→   235→}
   236→   236→
   237→   237→function subscribe(listener: () => void): () => void {
   238→   238→  listeners.add(listener);
   239→   239→  return () => {
   240→   240→    listeners.delete(listener);
   241→   241→  };
   242→   242→}
   243→   243→
   244→   244→function updateQuizState(next: QuizState): void {
   245→   245→  memoryState = next;
   246→   246→  try {
   247→   247→    if (next.stage === "intro") {
   248→   248→      sessionStorage.removeItem(STORAGE_KEY);
   249→   249→    } else {
   250→   250→      /* mantém quiz em andamento E resultado concluído salvos
   251→   251→         durante a sessão (recarregar não perde o progresso) */
   252→   252→      sessionStorage.setItem(STORAGE_KEY, JSON.stringify(next));
   253→   253→    }
   254→   254→  } catch {
   255→   255→    /* armazenamento indisponível: mantém apenas em memória */
   256→   256→  }
   257→   257→  listeners.forEach((listener) => listener());
   258→   258→}

   420→   420→  function handleListKeyDown(event: React.KeyboardEvent<HTMLDivElement>) {
   421→   421→    const deltas: Record<string, number> = {
   422→   422→      ArrowDown: 1,
   423→   423→      ArrowUp: -1,

   431→   431→    const focusedIndex = refs.findIndex((r) => r === document.activeElement);
